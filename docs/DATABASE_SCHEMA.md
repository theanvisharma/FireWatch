# FireWatch AI — Database Schema Specification
**Database:** PostgreSQL 16+ with PostGIS 3.4+ Extension  
**Spatial Reference System (SRID):** EPSG:4326 (WGS 84)

---

## 1. Extension Initialization
```sql
CREATE EXTENSION IF NOT EXISTS postgis;
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
```

---

## 2. Table Definitions

### Table: `monitored_locations`
Stores monitored geographic sectors and nature reserves.
```sql
CREATE TABLE monitored_locations (
    id VARCHAR(50) PRIMARY KEY,
    name VARCHAR(120) NOT NULL,
    region VARCHAR(120) NOT NULL,
    country VARCHAR(60) DEFAULT 'India',
    latitude DOUBLE PRECISION NOT NULL,
    longitude DOUBLE PRECISION NOT NULL,
    geom GEOMETRY(Point, 4326),
    boundary_geom GEOMETRY(Polygon, 4326),
    elevation_meters DOUBLE PRECISION,
    vegetation_type VARCHAR(100),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_locations_geom ON monitored_locations USING GIST(geom);
CREATE INDEX idx_locations_region ON monitored_locations(region);
```

### Table: `risk_predictions`
Stores time-stamped wildfire risk probability outputs generated by the AI engine.
```sql
CREATE TABLE risk_predictions (
    id VARCHAR(50) PRIMARY KEY DEFAULT uuid_generate_v4(),
    location_id VARCHAR(50) REFERENCES monitored_locations(id) ON DELETE CASCADE,
    risk_score DOUBLE PRECISION NOT NULL CHECK (risk_score >= 0 AND risk_score <= 100),
    risk_level VARCHAR(20) NOT NULL, -- Low, Moderate, High, Very High, Extreme
    prediction_horizon VARCHAR(50) DEFAULT '24-72 hours',
    factors JSONB NOT NULL, -- { temperature, wind_speed, rainfall, soil_moisture, ndvi, vpd }
    confidence DOUBLE PRECISION DEFAULT 0.90,
    model_version VARCHAR(30) DEFAULT 'v1',
    is_simulation BOOLEAN DEFAULT TRUE,
    generated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_predictions_loc_time ON risk_predictions(location_id, generated_at DESC);
CREATE INDEX idx_predictions_risk_level ON risk_predictions(risk_level);
```

### Table: `wildfire_alerts`
Early warnings triggered when risk scores cross critical thresholds.
```sql
CREATE TABLE wildfire_alerts (
    id VARCHAR(50) PRIMARY KEY,
    location_id VARCHAR(50) REFERENCES monitored_locations(id),
    severity VARCHAR(20) NOT NULL,
    risk_score DOUBLE PRECISION NOT NULL,
    message TEXT NOT NULL,
    preventive_action TEXT NOT NULL,
    is_read BOOLEAN DEFAULT FALSE,
    status VARCHAR(20) DEFAULT 'active', -- active, acknowledged, mitigated
    affected_radius_km DOUBLE PRECISION DEFAULT 15.0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_alerts_active ON wildfire_alerts(status, is_read);
```

### Table: `weather_observations`
Atmospheric and soil sensor readings per location.
```sql
CREATE TABLE weather_observations (
    id VARCHAR(50) PRIMARY KEY DEFAULT uuid_generate_v4(),
    location_id VARCHAR(50) REFERENCES monitored_locations(id),
    temperature DOUBLE PRECISION NOT NULL,
    humidity DOUBLE PRECISION NOT NULL,
    wind_speed DOUBLE PRECISION NOT NULL,
    wind_direction VARCHAR(20),
    rainfall_past_24h DOUBLE PRECISION DEFAULT 0.0,
    soil_moisture DOUBLE PRECISION NOT NULL,
    solar_radiation DOUBLE PRECISION,
    vpd_kpa DOUBLE PRECISION,
    recorded_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_weather_loc_time ON weather_observations(location_id, recorded_at DESC);
```

### Table: `historical_wildfires`
Ground-truth historical fire ignition events for training and trend analysis.
```sql
CREATE TABLE historical_wildfires (
    id VARCHAR(50) PRIMARY KEY,
    incident_date DATE NOT NULL,
    location_id VARCHAR(50),
    location_name VARCHAR(120) NOT NULL,
    region VARCHAR(120) NOT NULL,
    latitude DOUBLE PRECISION NOT NULL,
    longitude DOUBLE PRECISION NOT NULL,
    geom GEOMETRY(Point, 4326),
    max_risk_score DOUBLE PRECISION NOT NULL,
    severity VARCHAR(20) NOT NULL,
    area_burned_hectares DOUBLE PRECISION NOT NULL,
    containment_days INT DEFAULT 1,
    cause VARCHAR(60) NOT NULL,
    temperature_at_ignition DOUBLE PRECISION NOT NULL,
    wind_speed_at_ignition DOUBLE PRECISION NOT NULL,
    humidity_at_ignition DOUBLE PRECISION NOT NULL,
    soil_moisture_at_ignition DOUBLE PRECISION NOT NULL,
    satellite_detection_sensor VARCHAR(50) DEFAULT 'VIIRS'
);

CREATE INDEX idx_historical_geom ON historical_wildfires USING GIST(geom);
CREATE INDEX idx_historical_date ON historical_wildfires(incident_date DESC);
```
